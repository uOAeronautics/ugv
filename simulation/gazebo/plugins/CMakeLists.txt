cmake_minimum_required(VERSION 3.22)
project(ugv_tt_motor LANGUAGES CXX)
find_package(gz-sim10 QUIET)
if(gz-sim10_FOUND)
  set(SIM_TARGET gz-sim10::gz-sim10)
  find_package(gz-plugin4 REQUIRED COMPONENTS register)
  set(PLUGIN_TARGET gz-plugin4::register)
else()
  # Gazebo Harmonic (Sim 8) exports versioned package names.
  find_package(gz-sim8 QUIET)
endif()
if(gz-sim10_FOUND)
elseif(gz-sim8_FOUND)
  set(SIM_TARGET gz-sim8::gz-sim8)
  find_package(gz-plugin2 REQUIRED COMPONENTS register)
  set(PLUGIN_TARGET gz-plugin2::register)
else()
  find_package(gz-sim REQUIRED)
  find_package(gz-plugin REQUIRED COMPONENTS register)
  set(SIM_TARGET gz-sim::gz-sim)
  set(PLUGIN_TARGET gz-plugin::register)
endif()
add_library(ugv_tt_motor SHARED TTMotorSystem.cc)
target_compile_features(ugv_tt_motor PRIVATE cxx_std_17)
target_link_libraries(ugv_tt_motor PRIVATE ${SIM_TARGET} ${PLUGIN_TARGET})
if(UNIX AND NOT APPLE)
  # gz-sim exports GUI/AV/XML link dependencies the plugin never uses; on mixed
  # installs these can pull a second, ABI-incompatible tinyxml2/FFmpeg.
  target_link_options(ugv_tt_motor PRIVATE "LINKER:--as-needed")
endif()
